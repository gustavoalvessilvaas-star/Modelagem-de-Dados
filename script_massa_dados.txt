-- Verifica consultas realizadas sem evolucao (resultado esperado: 0 linhas).
SELECT c.id_consulta
FROM consulta c
LEFT JOIN evolucao_prontuario ev ON ev.id_consulta = c.id_consulta
WHERE c.status = 'realizada' AND ev.id_evolucao IS NULL;

-- Verifica exames concluidos sem resultado (resultado esperado: 0 linhas).
SELECT e.id_exame
FROM exame e
LEFT JOIN resultado_exame r ON r.id_exame = e.id_exame
WHERE e.status = 'concluido' AND r.id_resultado IS NULL;

-- Verifica convênios vinculados a consultas fora da vigência (resultado esperado: 0 linhas).
SELECT c.id_consulta
FROM consulta c
JOIN paciente_convenio pc
  ON pc.id_paciente_convenio = c.id_paciente_convenio
JOIN agenda a ON a.id_horario = c.id_horario
WHERE c.id_paciente_convenio IS NOT NULL
  AND (pc.ativo = FALSE OR a.data NOT BETWEEN pc.data_inicio AND pc.data_validade);
